import os
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from google import genai
import json
import random 
from pydantic import BaseModel
from typing import List
from dotenv import load_dotenv

class CasoDeTeste(BaseModel):
    id: str
    descricao: str
    classe_equivalencia: str
    dados_entrada: str
    resultado_esperado: str
    tipo: str # "Positivo" ou "Negativo"

class ListaTestes(BaseModel):
    variaveis_identificadas: List[str]
    casos_de_teste: List[CasoDeTeste]

load_dotenv()
client = genai.Client(api_key=os.getenv("GOOGLE_API_KEY"))
app = FastAPI()

app.add_middleware(
    CORSMiddleware,
    allow_origins=["http://localhost:3000"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

class EstoriaRequest(BaseModel):
    estoria: str

@app.post("/api/generate-tests")
async def generate_tests(request: EstoriaRequest):
    try:
        # 1. Validação de segurança imediata (Garante que não processe lixo)
        conteudo_limpo = request.estoria.strip()
        if len(conteudo_limpo) < 10:
            return {
                "variaveis_identificadas": [],
                "casos_de_teste": [],
                "erro_validacao": "A estória fornecida é muito curta ou irrelevante para gerar casos de teste técnicos."
            }

        # 2. Prompt com reforço de estrutura JSON
        prompt_engenharia = f"""
        Você é um engenheiro de QA especialista. Analise a estória: "{conteudo_limpo}"
        
        Siga estas regras:
        - Se o texto for irrelevante ou sem sentido funcional, retorne as listas vazias e preencha o campo 'erro_validacao'.
        - Se for válido, aplique Particionamento de Equivalência.
        
        Responda ESTRITAMENTE com o JSON no formato abaixo, sem textos explicativos:
        {{
            "variaveis_identificadas": ["lista"],
            "casos_de_teste": [
                {{
                    "id": "CT01",
                    "descricao": "string",
                    "classe_equivalencia": "string",
                    "dados_entrada": "string",
                    "resultado_esperado": "string",
                    "tipo": "Positivo ou Negativo"
                }}
            ],
            "erro_validacao": ""
        }}
        """

        response = client.models.generate_content(
            model="gemma-3-27b-it",
            contents=prompt_engenharia
        )

        # 3. Tratamento de limpeza de Markdown da IA
        texto_resposta = response.text.replace("```json", "").replace("```", "").strip()
        
        try:
            dados_saida = json.loads(texto_resposta)
            
            # Garantia de chaves: se a IA esquecer alguma, o Python injeta para não quebrar o Front
            if "variaveis_identificadas" not in dados_saida: dados_saida["variaveis_identificadas"] = []
            if "casos_de_teste" not in dados_saida: dados_saida["casos_de_teste"] = []
            
            return dados_saida

        except json.JSONDecodeError:
            # Fallback caso a IA falhe na formatação do JSON
            return {
                "variaveis_identificadas": [],
                "casos_de_teste": [],
                "erro_validacao": "Erro de processamento da IA. Por favor, tente enviar a estória novamente."
            }

    except Exception as e:
        print(f"Erro Crítico no Servidor: {e}")
        raise HTTPException(status_code=500, detail="Falha na comunicação com o modelo de IA.")
